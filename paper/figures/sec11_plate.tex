% (a) Decomposition of the plate's velocity relative to the air and the
% resulting drag, eq:screendrag. (b), (c) The moment of the screen drag about
% the centre of mass in forward flight, display above and below the rotors.
\begin{tikzpicture}[>={Stealth[length=2mm]}, font=\footnotesize]
  % ------------------------------------------------------------- (a) plate
  \begin{scope}
    \filldraw[fill=cE!25, draw=cE, line width=0.9pt] (-0.06,-1.3) rectangle (0.06,1.3);
    \node[cE, anchor=south] at (0,1.32) {screen, area $A_s$};
    \draw[vec, cE, line width=1pt] (0.08,0.55) -- (1.0,0.55) node[above] {$\vect n$};
    % velocity of the plate relative to the air
    \coordinate (P) at (0.1,0);
    \draw[vec, black, line width=1.1pt] (P) -- (2.3,-1.15);
    \node[anchor=north east] at (1.5,-0.75) {$\vect v_{\mathrm{rel}}$};
    \draw[vec, cB, line width=0.9pt] (P) -- (2.3,0);
    \node[cB, anchor=south] at (1.75,0.0) {$v_n\vect n$};
    \draw[vec, cB, line width=0.9pt] (2.3,0) -- (2.3,-1.15);
    \node[cB, anchor=west] at (2.32,-0.6) {$\vect v_t$};
    % drag: opposes each component, mostly the normal one
    \draw[vec, cD, line width=1.2pt] (-0.08,0) -- (-1.35,0.12);
    \node[cD, anchor=south] at (-0.85,0.15) {$\vect F_{\mathrm{screen}}$};
    \node[anchor=north, align=center] at (0.5,-1.45)
      {$\vect F_{\mathrm{screen}}=-\tfrac12\rho A_s\big[C_{D,n}|v_n|v_n\vect n$\\
       $\qquad\qquad+\,C_{D,t}\|\vect v_t\|\vect v_t\big]$};
    \node[anchor=north] at (0.5,-2.35) {(a)};
  \end{scope}
  % ------------------------------------------------- (b), (c) above / below
  % Tilted clockwise as drawn (b3 toward the motion). Drag at a point above
  % the centre of mass turns the body counterclockwise, against the tilt;
  % below, clockwise, with it.
  \foreach \dx/\up/\col/\txt/\lab in {5.6/1/cC/{opposes the tilt:\\weathercock-stable}/(b),
                                       9.8/-1/cD/{reinforces the tilt}/(c)} {
    \begin{scope}[shift={(\dx,0)}]
      \begin{scope}[rotate=-14]
        \draw[line width=1.6pt, black!55] (-1.3,0) -- (1.3,0);
        \foreach \x in {-1.3,1.3}
          \draw[cA, line width=0.9pt] (\x,0.03) ellipse (0.5 and 0.05);
        \filldraw[fill=cE!25, draw=cE, line width=0.8pt]
          (-0.05,{\up*0.55-0.45}) rectangle (0.05,{\up*0.55+0.45});
        \coordinate (S) at (0,{\up*0.55});
      \end{scope}
      \fill[black] (0,0) circle (0.07);
      \draw[vec, cD, line width=1.1pt] (S) -- ++(-1.0,0);
      \ifnum\up=1
        \draw[\col, line width=0.9pt, ->] (-145:0.62) arc (-145:-35:0.62);
      \else
        \draw[\col, line width=0.9pt, ->] (145:0.62) arc (145:35:0.62);
      \fi
      \draw[vec, cC] (0.9,-1.35) -- (1.7,-1.35) node[right] {$\vect v$};
      \node[\col, align=center, anchor=north] at (0,-1.55) {\txt};
      \node[anchor=north] at (0,-2.35) {\lab};
    \end{scope}
  }
  \node[anchor=south] at (5.6,1.45) {display above: $d_{s,3}>0$};
  \node[anchor=south] at (9.8,1.45) {display below: $d_{s,3}<0$};
\end{tikzpicture}
