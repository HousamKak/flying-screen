  \draw[line width=0.8pt] (0,-0.75) -- (0,0.75);

  \filldraw[fill=cA!12, draw=cA, line width=0.7pt] (0,-0.1) rectangle (6,0.1);

  \draw[cA!70, line width=1.2pt] (5.35,0.24) -- (6.65,0.24);
  \draw[black!60, line width=1pt] (6,0.1) -- (6,0.24);

  \draw[construct] (6,0.42) -- (6.9,0.42);
  \draw[construct] (6.7,0) -- (6.9,0);

    \node[anchor=west] at (3.5,-1.25) {root: $\sigma=\dfrac{M r}{I}\le\sigma_{\mathrm{all}}

  % cross-section
  \begin{scope}[shift={(8.9,-0.35)}]
    \fill[cA!25, even odd rule] (0,0) circle (0.95) (0,0) circle (0.78);

    \node[align=center, anchor=north] at (0,-1.35)
      {$I=\pi r^{3}t$,\quad $\mu=2\pi r t\rho_m$};
  \end{scope}
  \node[align=left, anchor=north west] at (-0.35,-2.75)
    {wall used: $t=\max\{t_\sigma,\ t_\delta,\ t_{\min}\}$};
